// Starter kerning set (ART_BIBLE §12.5). A pair kerns by -1 only when that leaves no ink of the
// two glyphs side by side or touching (forge pytest checks this). AW, WA, PA and ££ would touch at
// -1 with these draft shapes, so they stay at 0 until the M4 glyph pass (SSP-142) reshapes them.
kern "AV" -1
kern "AW" 0
kern "AT" -1
kern "AY" -1
kern "LT" -1
kern "LY" -1
kern "PA" 0
kern "TA" -1
kern "VA" -1
kern "WA" 0
kern "YA" -1
kern "Fo" -1
kern "To" -1
kern "Te" -1
kern "Ty" -1
kern "r." -1
kern "r," -1
kern "££" 0
